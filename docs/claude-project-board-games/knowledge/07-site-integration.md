# دمج لعبة جديدة في الموقع (مولَّد 2026-10-21)
خطوات إضافة لعبة إلى قسم «ألعاب السبورة الذكية» `/board-games` في منصة مناهج قطر (Next.js):

1. **مجلد اللعبة:** `public/<slug>/` (`index.html`، `js/main.js`، `js/game.js`، `js/content.js`، `css/game.css`،
   و`cover.jpg` لقطة 1280×800 من اللعبة أثناء اللعب).
2. **التحويل:** في `next.config.ts` داخل `redirects()` أضف:
   `{ source: '/<slug>', destination: '/<slug>/index.html', permanent: false }` والسطر نفسه مع `'/<slug>/'`.
3. **الفهرس:** أضف كائن اللعبة إلى `BOARD_GAMES` في `src/data/board-games.ts` (الحقول موثّقة في الملف أدناه).
4. **التجربة:** `/<slug>/index.html?sim` (الماوس يتصرف كيد)، ثم شاشة هاتف أفقية، ثم الجهاز الحقيقي في الصف.
5. **لا تكرّر** `vendor/` ولا `models/` ولا ملفات المحرّك داخل مجلد اللعبة: كلها في `/board-kit/`.

الملفات الحالية للمرجع:

// ---------------------------------------------------------------------------
// فهرس «ألعاب السبورة الذكية»: ألعاب تُلعب بلمس السبورة باليد، يعرضها جهاز
// (آيباد أو هاتف) على جهاز العرض عبر Apple TV ويرى أيدي الطلاب بكاميرته.
//
// كل لعبة تطبيق ويب ثابت في public/<slug>/ مبني على المحرّك المشترك
// public/board-kit/ (راجع board-kit/README_AR.md). لإضافة لعبة: أنشئ مجلدها،
// وأضف تحويل /<slug> في next.config.ts، ثم أضف سطرها هنا — فتظهر في القسم.
// ---------------------------------------------------------------------------

export interface BoardGame {
  /** اسم المجلد في public/ وأساس الرابط */
  slug: string;
  title: string;
  /** وصف قصير يظهر في البطاقة (سطران) */
  summary: string;
  subject: string;
  /** الصف/المستوى والوحدة أو الدرس */
  context?: string;
  /** نواتج التعلّم الرئيسية (٢–٤ نقاط قصيرة) */
  objectives: string[];
  /** أوضاع اللعب كما تظهر في قائمة اللعبة */
  modes: string[];
  /** فردي/الصف كله/سباق فريقين… */
  players: string;
  /** مدة الجولة التقريبية بالدقائق */
  minutes: number;
  /** صورة الغلاف (لقطة من اللعبة، نسبة 16:10) */
  cover: string;
  /** لون البطاقة */
  color: string;
  /** تاريخ الإضافة (YYYY-MM-DD) — الأحدث يظهر أولًا */
  added: string;
}

}
